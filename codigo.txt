function calcular(){
  const d=loadDB();const cfg=d.configs;
  const largura=parseNum(document.getElementById('largura').value);
  const altura=parseNum(document.getElementById('altura').value);
  const gap=parseNum(document.getElementById('gap').value);
  const qtd=parseNum(document.getElementById('quantidade').value);
  const matNome=document.getElementById('material').value;
  const vendNome=document.getElementById('vendedor-select').value;
  const areaUtil=cfg.areaUtilImpressao||0.9;

  const modoEl=document.querySelector('input[name="modo"]:checked');
  const modoKey=modoEl?modoEl.value:'Padrao';
  const modo=MODOS.find(m=>m.key===modoKey)||MODOS[1];
  const custoModo=cfg[modo.custo];
  const velModo=cfg[modo.vel]||10;

  const coresCMYK=[];
  if(document.getElementById('cor-yellow').checked)coresCMYK.push('Yellow');
  if(document.getElementById('cor-magenta').checked)coresCMYK.push('Magenta');
  if(document.getElementById('cor-cyan').checked)coresCMYK.push('Cyan');
  if(document.getElementById('cor-black').checked)coresCMYK.push('Black');
  const custoCMYK=custoModo*(coresCMYK.length/4);

  let custoWhite=0,whiteCov=0;
  if(document.getElementById('cor-white').checked){
    whiteCov=parseNum(document.getElementById('cor-branco-cov').value)||20;
    custoWhite=cfg.whiteChapada*(whiteCov/100);
  }
  const canais=[...coresCMYK];
  if(custoWhite>0)canais.push('White '+whiteCov+'%');

  const pctVend=parseNum(document.getElementById('vendedor').value)||0;
  const pctEmp=parseNum(document.getElementById('empresa').value)||0;
  const frete=parseNum(document.getElementById('frete').value)||0;

  if([largura,altura,gap,qtd].some(v=>isNaN(v))||qtd<=0){toast('⚠️ Preencha dimensões e quantidade',true);return;}
  const mat=d.materiais.find(m=>m.nome===matNome);
  if(!mat){toast('⚠️ Material não encontrado',true);return;}
  if(canais.length===0){toast('⚠️ Selecione ao menos uma cor',true);return;}

  const lB=(largura+2*gap)/1000,aB=(altura+2*gap)/1000;
  const areaU=lB*aB,areaL=areaU*qtd,areaP=areaL*(1+cfg.perda/100);
  const etiquetasM2=areaU>0?Math.floor(areaUtil/areaU):0;

  const cMat=mat.valor*(1+mat.icms/100);
  const subTinta=custoCMYK+custoWhite;
  const valPurga=subTinta*(cfg.purga/100);
  const cTin=subTinta+valPurga;
  const custoVarM2=cMat+cTin;
  const custoVarTotal=custoVarM2*areaP;

  const tempoProducao=areaP/velModo;
  const custoMaquinaTotal=tempoProducao*cfg.horaMaquina;

  const cTir=custoVarTotal+custoMaquinaTotal;
  const cUni=cTir/qtd;
  const vEmp=cTir*pctEmp/100;const vVend=cTir*pctVend/100;
  const pTot=cTir+vEmp+vVend+frete;const pUni=pTot/qtd;

  ultimoResultado={largura,altura,gap,qtd,matNome,canais,modo:modo.label+' ('+modo.res+')',modoKey,velModo,areaP,tempoProducao,custoMaquinaTotal,etiquetasM2,areaUtil,cTin,cTir,vEmp,vVend,frete,pTot,pUni,pctEmp,pctVend,vendNome};

  salvarNoHistorico({id:Date.now(),data:new Date().toLocaleDateString('pt-BR')+' '+new Date().toLocaleTimeString('pt-BR',{hour:'2-digit',minute:'2-digit'}),dataBr:new Date().toLocaleDateString('pt-BR'),vendedor:vendNome||'—',largura,altura,gap,quantidade:qtd,material:matNome,modo:modo.label,modoKey,canais,areaPerda:areaP,tempoProducao:tempoProducao,custoMaquina:custoMaquinaTotal,precoTotal:pTot,precoUnit:pUni,pctVendedor:pctVend,pctEmpresa:pctEmp,frete,custoTiragem:cTir,valEmpresa:vEmp,valVendedor:vVend});

  function ri(l,v){return'<div class="r-item"><span>'+l+'</span><b>'+v+'</b></div>';}
  let h='<div class="hero"><div class="t">💰 Preço de Venda Total</div><div class="v">'+brl(pTot)+'</div><div class="u">'+brl(pUni,4)+'/unid · '+Math.round(qtd).toLocaleString('pt-BR')+' un'+(vendNome?' · '+vendNome:'')+'</div></div>';

  h+='<div class="r-sec">Áreas</div>'+ri('Bruta',(lB*1000).toFixed(0)+'×'+(aB*1000).toFixed(0)+' mm')+ri('Área Unidade',areaU.toFixed(4)+' m²')+ri('Rendimento (área útil '+areaUtil.toFixed(1)+' m²)',etiquetasM2+' etiquetas/m²')+ri('Área Líquida',areaL.toFixed(3)+' m²')+ri('Área c/ Perda ('+cfg.perda+'%)',areaP.toFixed(3)+' m²');

  h+='<div class="r-sec">Impressão — '+modo.label+'</div>';
  h+=ri('Resolução',modo.res);
  h+=ri('Velocidade do modo',velModo+' m²/h');
  h+=ri('Canais ativos',canais.join(', '));

  h+='<div class="r-sec">Custos Variáveis /m²</div>';
  h+=ri('Material ('+matNome+')',brl(cMat,3));
  h+=ri('· Tinta CMYK ('+coresCMYK.length+'/4)',brl(custoCMYK,3));
  if(custoWhite>0)h+=ri('· Tinta White ('+whiteCov+'%)',brl(custoWhite,3));
  h+=ri('· Purga ('+cfg.purga+'%)','+ '+brl(valPurga,3));
  h+=ri('Subtotal Variável /m²',brl(custoVarM2,3));

  h+='<div class="r-sec">⏱️ Produção & Máquina</div>';
  h+=ri('Tempo de produção',areaP.toFixed(3)+' ÷ '+velModo+' m²/h = <b>'+fmtTempo(tempoProducao)+'</b>');
  h+=ri('Custo da máquina',fmtTempo(tempoProducao)+' × '+brl(cfg.horaMaquina)+'/h');
  h+='<div class="r-total"><span>CUSTO MÁQUINA</span><span>'+brl(custoMaquinaTotal)+'</span></div>';

  h+='<div class="r-sec">💵 Custos Totais</div>';
  h+=ri('Custos variáveis',brl(custoVarTotal));
  h+=ri('Custo máquina',brl(custoMaquinaTotal));
  h+='<div class="r-total"><span>CUSTO TIRAGEM</span><span>'+brl(cTir)+'</span></div>';
  h+=ri('Custo / Rótulo',brl(cUni,4));

  h+='<div class="r-sec">Comercial</div>';
  h+='<div class="r-total comm"><span>Empresa ('+pctEmp+'%)</span><b>+ '+brl(vEmp)+'</b></div>';
  h+='<div class="r-total comm"><span>Vendedor ('+pctVend+'%)</span><b>+ '+brl(vVend)+'</b></div>';
  h+='<div class="r-total comm"><span>Frete</span><b>+ '+brl(frete)+'</b></div>';
  h+='<button class="btn-pdf" onclick="gerarPDF()">📄 BAIXAR PDF</button>';
  h+='<button class="btn-share" onclick="compartilhar()">📤 COMPARTILHAR</button>';
  h+='<div class="sheet-ds">DS · Daniel Souza</div>';
  document.getElementById('sheet-body').innerHTML=h;
  document.getElementById('sheet-overlay').classList.add('open');
  document.body.style.overflow='hidden';
  if(navigator.vibrate)navigator.vibrate(30);
}
